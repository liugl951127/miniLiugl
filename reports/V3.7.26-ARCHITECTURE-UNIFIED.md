## 1. 后端 SseResult 工具类

`backend/minimax-common/src/main/java/com/minimax/common/sse/SseResult.java`：

```java
public static void send(SseEmitter emitter, String event, Object data) {
  // 自动包 Result: {code:0, message:"success", data, timestamp}
  Map<String, Object> payload = new LinkedHashMap<>();
  payload.put("code", 0);
  payload.put("message", "success");
  payload.put("data", data);
  payload.put("timestamp", System.currentTimeMillis());
  emitter.send(SseEmitter.event().name(event).data(payload, MediaType.APPLICATION_JSON));
}

public static void sendError(SseEmitter emitter, String errorMessage) {
  // code=1, data:{message: errorMessage}
  Map<String, Object> errData = Map.of("message", errorMessage);
  send(emitter, "error", errData, 1, errorMessage);
}

public static void sendDone(SseEmitter emitter) {
  send(emitter, "done", Map.of("status", "finished"));
}
```

## 2. 后端 chat MessageController 真正 SseEmitter

```java
@PostMapping(value = "/stream", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
public SseEmitter streamMessage(...) {
  SseEmitter emitter = new SseEmitter(120_000L);
  SSE_EXECUTOR.execute(() -> {
    try {
      // 1. start
      SseResult.send(emitter, "start", Map.of("streamId", UUID.randomUUID().toString()));
      
      // 2. content chunks
      for (String word : words) {
        SseResult.send(emitter, "content", Map.of("content", word));
        Thread.sleep(50);
      }
      
      // 3. done
      SseResult.sendDone(emitter);
    } catch (Exception e) {
      SseResult.sendError(emitter, e.getMessage());
    } finally {
      SseResult.complete(emitter);
    }
  });
  return emitter;
}
```

export async function sendMessageStream(sessionId, body, opts = {}) {
  const stream = useBusinessStream()
  return stream.send(`/sessions/${sessionId}/messages/stream`, 
    { ...body, streamId: opts.streamId },
    {
      onContent: opts.onContent || opts.onChunk,  // 兼容
      onToolCall: opts.onToolCall,
      onSource: opts.onSource,
      onDone: opts.onDone,
      onError: opts.onError,
      signal: opts.signal,
    }
  )
}

## 4. 前端 useChatStream 委托

```js
// 之前: 重复 90 行实现
// V3.7.26+: 委托 useBusinessStream + 兼容 onChunk 别名
export function useChatStream() {
  const stream = useBusinessStream()
  const send = async (url, payload, callbacks = {}) => {
    const { onChunk, onContent, ...rest } = callbacks
    return stream.send(url, payload, {
      ...rest,
      onContent: onContent || onChunk,
    })
  }
  return { ...stream, send }
}
```
